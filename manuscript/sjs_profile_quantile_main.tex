\title{Debiased Profile Inference for High-Dimensional Mixed-Effects Quantile Regression with Many Small Clusters}

\author[1]{Jian Hou}
\author[1]{Tan Meng}
\author[1]{Maozai Tian}

\authormark{Hou et al.}
\titlemark{Debiased profile inference for mixed-effects quantile regression}

\address[1]{\orgdiv{School of Statistics}, \orgname{Renmin University of China}, \orgaddress{\city{Beijing}, \country{China}}}

\corres{Correspondence: Jian Hou, School of Statistics, Renmin University of China, Beijing, China. Email: houjian@ruc.edu.cn}

\abstract[Abstract]{High-dimensional longitudinal studies combine sparse fixed effects, within-cluster dependence, and a large collection of cluster-specific nuisance parameters. We develop a regularised profile framework for quantile regression when the number of independent clusters diverges, cluster sizes remain bounded, and the dimension of the fixed-effect vector can exceed the number of clusters. The inferential target is an unsmoothed ridge-profile population parameter, so the target does not depend on the computational bandwidth. Convolution smoothing is used to compute the estimator and estimate local curvature. Profiling the complete ridge-regularised cluster criterion yields an exact score based on the original fixed-effect design and a Schur-complement Hessian containing an additional ridge-curvature term. Row-wise Dantzig inversion of this effective Hessian and a one-step correction produce coordinate-wise debiased estimators. We formulate target-approximation and estimation bounds, a cluster-level Bahadur representation, asymptotic normality, and a consistent sandwich variance estimator under a non-empty bandwidth--sparsity regime. The numerical design evaluates estimation and inference without empirical coverage calibration. The proposed application uses the longitudinal GSE65391 paediatric systemic lupus erythematosus study, with continuous complement C3 as the response and whole-blood expression profiles as high-dimensional predictors.}

\keywords{clustered data; convolution smoothing; debiased inference; high-dimensional statistics; mixed-effects model; profile M-estimation; quantile regression}
