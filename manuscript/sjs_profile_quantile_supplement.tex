\title{Supporting Information for Joint Profile Inference for High Dimensional Mixed Effects Quantile Regression with Many Small Clusters}
\author[1]{Jian Hou}\author[1]{Tan Meng}\author[1]{Maozai Tian}
\authormark{Hou et al.}\titlemark{Supporting Information}
\address[1]{\orgdiv{School of Statistics}, \orgname{Renmin University of China}, \orgaddress{\city{Beijing}, \country{China}}}
\corres{Jian Hou. Email: houjian@ruc.edu.cn}
\abstract[Summary]{We give the complete profile calculus, joint-density bandwidth expansion and sparse empirical bounds. These results establish the learning rates of the orthogonal search and nodewise adjustment rules, control the selected-model target and yield paired coefficient inference. Numerical specifications include certified convex optimisation, the independent simulation and its paired ablation. The soybean field-trial analysis provides the observed panel construction, genomic adjustment, joint coordinate and contrast intervals, sensitivity calculations and held-out prediction assessment.}
\keywords{clustered data; joint optimisation; profile Hessian; quantile regression; supporting information}
